cmake_minimum_required(VERSION 3.22)
project(DeskPetNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML_ALLOW_FALLBACK OFF CACHE BOOL "" FORCE)
set(DESKPET_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.build/native-src" CACHE PATH "Native dependency checkouts")
set(LLAMA_SOURCE "${DESKPET_SOURCE_ROOT}/llama")
set(WHISPER_SOURCE "${DESKPET_SOURCE_ROOT}/whisper")
include(strict-gpu.cmake)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_library(deskpet_inference SHARED backend.cpp router.cpp speech.cpp)
target_include_directories(deskpet_inference PRIVATE "../Sources/CNativeInference/include" "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/vendor/nlohmann")
target_link_libraries(deskpet_inference PRIVATE llama ggml llama-common whisper)
set_target_properties(deskpet_inference PROPERTIES BUILD_RPATH "@loader_path;@loader_path/llama/src;@loader_path/llama/ggml/src;@loader_path/whisper/src")
option(DESKPET_NATIVE_TESTS "Build graph/template contracts without model inference" OFF)
if(DESKPET_NATIVE_TESTS)
  add_executable(deskpet-native-tests tests/router-contract.cpp)
  target_include_directories(deskpet-native-tests PRIVATE "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/vendor/nlohmann")
  target_link_libraries(deskpet-native-tests PRIVATE ggml llama-common)
endif()
set(SHERPA_ROOT "" CACHE PATH "sherpa-onnx installed C API and ONNX Runtime")
if(SHERPA_ROOT)
  find_library(SHERPA_LIBRARY sherpa-onnx-c-api PATHS "${SHERPA_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
  find_library(ORT_LIBRARY onnxruntime PATHS "${SHERPA_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
  target_compile_definitions(deskpet_inference PRIVATE DESKPET_SPEECH=1)
  target_include_directories(deskpet_inference PRIVATE "${SHERPA_ROOT}/include" "${SHERPA_ROOT}/include/onnxruntime/core/session")
  target_link_libraries(deskpet_inference PRIVATE "${SHERPA_LIBRARY}" "${ORT_LIBRARY}")
endif()
option(DESKPET_PREP "Build offline checkpoint preparation tool (LibTorch required)" OFF)
if(DESKPET_PREP)
  find_package(Torch REQUIRED)
  add_executable(deskpet-model-prep prepare.cpp)
  target_include_directories(deskpet-model-prep PRIVATE "${LLAMA_SOURCE}/vendor/nlohmann")
  target_link_libraries(deskpet-model-prep PRIVATE llama ggml "${TORCH_LIBRARIES}")
  if(SHERPA_ROOT)
    target_compile_definitions(deskpet-model-prep PRIVATE DESKPET_PREP_ORT=1)
    target_include_directories(deskpet-model-prep PRIVATE "${SHERPA_ROOT}/include/onnxruntime/core/session")
    target_link_libraries(deskpet-model-prep PRIVATE "${ORT_LIBRARY}")
  endif()
endif()
